from hub import port
import runloop, motor_pair

CM_PER_ROT = 17.6
def deg_for_cm(cm): return int(cm/ CM_PER_ROT * 360)

async def main():
    motor_pair.pair(motor_pair.PAIR_1, port.C, port.D)
    deg = deg_for_cm(40)

    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        deg,
        200,
        700
    )
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        deg,
        700,
        200
    )
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        deg,
        700,
        200
    )
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        deg,
        200,
        700
    )
runloop.run(main())
