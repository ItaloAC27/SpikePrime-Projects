from hub import port
import runloop, motor_pair
CM_PER_ROT = 17.6
def deg_for_cm(cm):
    return int(cm/CM_PER_ROT*360)
async def main():
    motor_pair.pair(motor_pair.PAIR_1, port.C, port.D)
    await motor_pair.move_for_degrees(
        motor_pair.PAIR_1,
        deg_for_cm(30),
        0,
        velocity=600
    )
    await motor_pair.move_for_degrees(
        motor_pair.PAIR_1,
        -deg_for_cm(30),
        0,
        velocity=600
    )
    TURN_DEG_MOT = 200
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        TURN_DEG_MOT,
        -600,
        +600
    )
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        TURN_DEG_MOT,
        +600,
        -600
    )
    deg = deg_for_cm(40)

    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        deg,
        200,
        700
    )
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        deg,
        700,
        200
    )
    deg = deg_for_cm(40)

    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        deg,
        200,
        700
    )
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        deg,
        700,
        200
    )
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        deg,
        700,
        200
    )
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        deg,
        200,
        700
    )
    for _ in range(4):
        await motor_pair.move_for_degrees(
            motor_pair.PAIR_1,
            deg_for_cm(30),
            0,
            velocity=1000
        )
        await motor_pair.move_tank_for_degrees(
            motor_pair.PAIR_1,
            180,
            600,
            -600
        )

runloop.run(main())
