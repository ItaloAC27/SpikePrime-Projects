GIRA ESQUERDA E DIREITA

from hub import port
import runloop, motor_pair
async def main():
    motor_pair.pair(motor_pair.PAIR_1, port.C, port.D)
    TURN_DEG_MOT = 200
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        TURN_DEG_MOT,
        -600,
        +600
    )
    await motor_pair.move_tank_for_degrees(
        motor_pair.PAIR_1,
        TURN_DEG_MOT,
        +600,
        -600
    )
    runloop.run(main())
